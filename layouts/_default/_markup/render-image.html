{{- /* 图片渲染钩子。 行为： 1) 若目标图片是"页面资源"（page bundle，如 ![图](photo.png)）， 走 Hugo Images Pipeline：生成多尺寸 WebP，输出 srcset/sizes/width/height （width/height 用于防止 CLS），同时保留原图作为灯箱放大源（data-zoom-src）。 2) 否则（static 目录或外链，如 /img/a.png、https://…）保持原有输出， 路径写法与旧版完全一致，不影响已有站点资源。 配置（可选）： [params.images] widths = [480, 768, 1200] sizes = "(max-width: 768px) 100vw, 900px" quality = 85 format = "webp" */ -}} {{- $dest := .Destination -}} {{- $alt := .Text | default (.Title | default "") -}} {{- $img := .Page.Resources.GetMatch $dest -}} {{- $page := .Page -}} {{- /* 可处理性判断：**不能用 reflect.IsImageResourceProcessable** —— 该函数在 Hugo 0.128.0 上不存在（实测报 "can't evaluate field IsImageResourceProcessable"）， 而 0.128.0 是 theme.toml 声明的最低版本。 这里改为按 MediaType 的 raster 白名单判断，所有支持版本行为一致。 */ -}} {{- $processable := false -}} {{- with $img -}} {{- if eq .ResourceType "image" -}} {{- $processable = in (slice "jpeg" "png" "webp" "gif" "bmp" "tiff") .MediaType.SubType -}} {{- end -}} {{- end -}} {{- if and $img $processable -}} {{- with $img -}} {{- $cfg := site.Params.images | default dict -}} {{- $widths := $cfg.widths | default (slice 480 768 1200) -}} {{- $quality := $cfg.quality | default 85 -}} {{- $format := $cfg.format | default "webp" -}} {{- $sizes := $cfg.sizes | default "(max-width: 768px) 100vw, 900px" -}} {{- $srcset := slice -}} {{- $fallback := .RelPermalink -}} {{- range $widths -}} {{- if ge $img.Width . -}} {{- $r := $img.Resize (printf "%dx %s q%d" . $format $quality) -}} {{- $srcset = $srcset | append (printf "%s %dw" $r.RelPermalink $r.Width) -}} {{- end -}} {{- end -}} {{- if not $srcset -}} {{- $r := $img.Resize (printf "%dx %s q%d" $img.Width $format $quality) -}} {{- $srcset = $srcset | append (printf "%s %dw" $r.RelPermalink $r.Width) -}} {{- end -}} {{- end -}} {{- else if $img -}} {{- /* SVG 等不可处理的图片资源：输出原图（保持原行为） */ -}} {{- else -}} {{- $src := partial "util/rel-url.html" $dest | safeURL -}} {{- end -}}
